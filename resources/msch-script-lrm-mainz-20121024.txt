### --- load data: msch-data-mainz-20121024
rm(list=ls(all=T))
mydata <-read.table(choose.files(), header=T, sep="\t", comment.char="", quote="")
attach(mydata); str(mydata); summary(mydata)

### --- Plot: eye-balling data ith association plots
par(mfrow=c(1,3))
assocplot(table(PAK, HIT))
assocplot(table(PRIMING, HIT))
spineplot (HIT ~ AGE)
par(mfrow=c(1,1))

### --- set options
mydata.dd = datadist(mydata)
#options(contrasts=c("contr.treatment", "contr.poly"))
options(datadist = "mydata.dd")
xtabs( ~ mydata$HIT)

### --- create saturated model
model.lrm = lrm(HIT ~ SEX + AGE + PAK + MAO + TYPE + UML + SML + CLE + MAN + PRO + SEX * PAK + SEX * MAO + SEX * AGE+ PRIMING, data = mydata, x = T, y = T)

### --- report: anova()
anova(model.lrm)

### --- report: model 
model.lrm

### --- model validation
validate(model.lrm, bw = T, B = 200)

#####################################################################
### --- model fitting process stepwise step down
### --- eliminate insignificant predictors

### --- see if model statistics improve after predictor

### --- new minimal model
model.lrm.2 = lrm(HIT ~ SEX + AGE + PAK + SEX * PAK + 
PRIMING, data = mydata, x = T, y = T)

### --- report: anova 
anova(model.lrm)

### --- report: model 
model.lrm

### --- model validation
validate(model.lrm, bw = T, B = 200)

### --- correct for overfitting 
### --- best penalty is with highest aic!
pentrace(model.lrm, seq(0, 0.95, by = 0.05))

#### ---- create new model with penalty=.95
model.lrm.pen = update(model.lrm, penalty = 0.95)

#### --- report new
model.lrm.pen

### --- report: anova
anova(model.lrm)

### --- model validation
validate(model.lrm.pen, bw = T, B = 200)

### --- check if penalty has shrunk coefficients (if yes = good)
cbind(coef(model.lrm), coef(model.lrm.pen),
abs(coef(model.lrm) - coef(model.lrm.pen)))

### --- check how good the predictions of the model are
vorhersagen<-predict.lrm(model.lrm.pen)
vorhersagen<-ifelse(vorhersagen<=0, 0,1)
auswertung<-table(vorhersagen, HIT)
auswertung
sum(diag(auswertung))/sum(auswertung)

### --- report effect size as odds ratios
oddsratios.lrm <- exp(1)^model.lrm.pen$coefficients
oddsratios.lrm
